301 /**
302 * @brief Enables or disables using the atmosphere-generated skybox as the
303 * scene skybox.
304 *
305 * When enabled, the renderer is expected to populate `atmosphereSkybox`,
306 * and the scene will expose it via `getSkybox()`.
307 *
308 * When disabled, the scene will use the user-provided skybox set through
309 * `setSkybox()`.
310 */
311 void setUseAtmosphereSkybox(bool enabled) {
312 useAtmosphereSkybox = enabled;
313 skybox = useAtmosphereSkybox ? atmosphereSkybox : userSkybox;
314 if (automaticAmbient) {
315 updateAutomaticAmbientFromSkybox();
316 }
317 }

390 Color automaticAmbientColor = Color::white();
391 float automaticAmbientIntensity = ambientLight.intensity;
392
393 void updateAutomaticAmbientFromSkybox() {
394 if (skybox != nullptr && skybox->cubemap.hasAverageColor) {
395 automaticAmbientColor = skybox->cubemap.averageColor;
396 double luminance = (0.2126 * automaticAmbientColor.r) +
397 (0.7152 * automaticAmbientColor.g) +
398 (0.0722 * automaticAmbientColor.b);
399 automaticAmbientIntensity =
400 static_cast<float>(std::clamp(luminance, 0.0, 1.0));
401 } else {
402 automaticAmbientColor = ambientLight.color;
403 automaticAmbientIntensity = ambientLight.intensity;
404 }

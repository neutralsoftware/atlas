68/**
69 * @brief Singleton class that manages all resources and resource groups in the
70 * application. Provides centralized access to assets like images, audio files,
71 * and other resources.
72 *
73 * \subsection workspace-example Example
74 * ```cpp
75 * // Get the workspace instance
76 * Workspace &workspace = Workspace::get();
77 * // Set the root path for resources
78 * workspace.setRootPath("assets/");
79 * // Create a resource
80 * Resource texture = workspace.createResource("textures/brick.png",
81 * "BrickTexture", ResourceType::Image);
82 * // Create a resource group
83 * std::vector<Resource> skyboxResources = {...};
84 * ResourceGroup skybox = workspace.createResourceGroup("Skybox",
85 * skyboxResources);
86 * ```
87 *
88 */
89class Workspace {
90 public:
91 static Workspace &get() {
92 static Workspace instance;
93 return instance;
94 }

69/**
70 * @brief Class representing a physical body in a physics simulation. It
71 * includes methods for applying forces, detecting collisions, and updating the
72 * body's state.
73 * \subsection body-example Example
74 * ```cpp
75 * // Create a physics body
76 * auto body = std::make_shared<Body>();
77 * // Set the body's position
78 * body->position = {0.0, 0.0, 0.0};
79 * // Apply a mass of 2.0 units
80 * body->applyMass(2.0f);
81 * // Set the body's shape to a sphere with radius 1.0
82 * body->shape = std::make_shared<Sphere>(1.0f);
83 * // Set the body's elasticity and friction
84 * body->elasticity = 0.8f;
85 * body->friction = 0.5f;
86 * ```
87 */
88class Body {
89 public:
90 /** @brief World-space position of the body. */
91 Position3d position;
92 /** @brief Orientation represented as a quaternion. */
93 glm::quat orientation;

168 /**
169 * @brief Applies mass to the body, setting the inverse mass.
170 *
171 * @param mass The mass to apply. If <= 0, creates infinite mass.
172 */
173 inline void applyMass(float mass) {
174 if (mass <= 0.0f) {
175 invMass = 0.0f; // Infinite mass
176 } else {
177 invMass = 1.0f / mass;
178 }
179 }
180
181 /**
182 * @brief Gets the mass of the body.
183 *
184 * @return (float) The mass of the body, or INFINITY for infinite mass.
185 */
186 inline float getMass() const {
187 if (invMass == 0.0f) {
188 return INFINITY;
189 }
190 return 1.0f / invMass;
191 }

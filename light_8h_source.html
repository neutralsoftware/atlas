576/**
577 * @brief Rectangular area light with controllable emission angle and two-sided
578 * emission. The rectangle is defined by its center (position) and two oriented
579 * axes (right, up) and size (width, height). The plane normal is
580 * normalize(cross(right, up)).
581 */
582struct AreaLight {
583 /**
584 * @brief Center position of the rectangle.
585 */
586 Position3d position = {0.0, 0.0, 0.0};
587 /**
588 * @brief Oriented axis for width direction (normalized).
589 */
590 Magnitude3d right = {1.0, 0.0, 0.0};
591 /**
592 * @brief Oriented axis for height direction (normalized).
593 */
594 Magnitude3d up = {0.0, 1.0, 0.0};
595 /**
596 * @brief Width and height of the rectangle.
597 */
598 Size2d size = {1.0, 1.0};
599
600 /**
601 * @brief Diffuse/emissive color of the light.
602 */
603 Color color = Color::white();
604 /**
605 * @brief Specular highlight color for the light.
606 */
607 Color shineColor = Color::white();
608 /**
609 * @brief Overall brightness multiplier applied to the area light.
610 */
611 float intensity = 1.0f;
612 /**
613 * @brief Maximum distance at which this area light contributes.
614 */
615 float range = 50.0f;
616
617 /**
618 * @brief Emission cone half-angle in degrees around the plane normal.
619 * For example, 90 means a hemisphere emission relative to the plane normal.
620 */
621 float angle = 90.0f;
622
623 /**
624 * @brief If true, the light emits on both sides of the rectangle plane.
625 */
626 bool castsBothSides = false;
627
628 /**
629 * @brief Rotation tracking for the area light. Changing this and calling
630 * setRotation/rotate will update right/up consistently.
631 *
632 * Rotation order matches CoreObject (roll Z, then pitch X, then yaw Y).
633 */
634 Rotation3d rotation = {0.0, 0.0, 0.0};
635
636 /**
637 * @brief Compute the plane normal (normalize(cross(right, up))).
638 */
639 Magnitude3d getNormal() const {
640 glm::vec3 u = glm::normalize(right.toGlm());
641 glm::vec3 v = glm::normalize(up.toGlm());
642 glm::vec3 n = glm::normalize(glm::cross(u, v));
643 return Magnitude3d::fromGlm(n);
644 }
645
646 /**
647 * @brief Convenience to set diffuse color.
648 */
649 void setColor(Color c) { color = c; }
650
651 /**
652 * @brief Sets absolute rotation (in degrees) and updates right/up
653 * accordingly. Rotation is applied to a canonical frame (right=+X, up=+Y,
654 * normal=+Z) in the order: roll(Z), pitch(X), yaw(Y).
655 */
656 void setRotation(const Rotation3d &r) {
657 rotation = r;
658 updateAxesFromRotation();
659 }
660
661 /**
662 * @brief Applies a delta rotation (in degrees) and updates right/up.
663 */
664 void rotate(const Rotation3d &delta) {
665 rotation = rotation + delta;
666 updateAxesFromRotation();
667 }
668
669 /**
670 * @brief Debug helpers for visualizing the rectangle in the scene.
671 */
672 void createDebugObject();
673 void addDebugObject(Window &window);
674 void castShadows(Window &window, int resolution = 2048);
675
676 /**
677 * @brief Optional debug object visualizing the area light.
678 */
679 std::shared_ptr<CoreObject> debugObject = nullptr;
680 /**
681 * @brief Optional shadow render target generated for this area light.
682 */
683 RenderTarget *shadowRenderTarget = nullptr;
684 /**
685 * @brief Cached shadow matrices and bias for the current area-light map.
686 */
687 ShadowParams lastShadowParams;
688
689 private:
690 bool doesCastShadows = false;
691 ShadowParams calculateLightSpaceMatrix() const;
692 /**
693 * @brief Recompute right/up from the current rotation to keep a coherent
694 * frame.
695 */
696 void updateAxesFromRotation() {
697 // Canonical basis
698 glm::vec3 baseRight(1.0f, 0.0f, 0.0f);
699 glm::vec3 baseUp(0.0f, 1.0f, 0.0f);
700 // Construct rotation matrix (roll Z, then pitch X, then yaw Y)
701 glm::mat4 m(1.0f);
702 m = glm::rotate(m, glm::radians(static_cast<float>(rotation.roll)),
703 glm::vec3(0, 0, 1));
704 m = glm::rotate(m, glm::radians(static_cast<float>(rotation.pitch)),
705 glm::vec3(1, 0, 0));
706 m = glm::rotate(m, glm::radians(static_cast<float>(rotation.yaw)),
707 glm::vec3(0, 1, 0));
708
709 glm::vec3 R = glm::normalize(glm::vec3(m * glm::vec4(baseRight, 0.0f)));
710 glm::vec3 U = glm::normalize(glm::vec3(m * glm::vec4(baseUp, 0.0f)));
711 // Orthonormalize to avoid drift
712 glm::vec3 N = glm::normalize(glm::cross(R, U));
713 R = glm::normalize(glm::cross(U, N));
714 U = glm::normalize(glm::cross(N, R));
715
716 right = Magnitude3d::fromGlm(R);
717 up = Magnitude3d::fromGlm(U);
718 }
